SELECT

AGGREGAT
#
Wie hoch waren die Wartungskosten insgesamt und im Durchschnitt (gerundet auf zwei Nachkommastellen)?
#
SUM() und AVG() lassen sich in einer Abfrage kombinieren.

#
JOIN
#
Gib alle Mitarbeitern mit dem Namen ihrer Abteilung aus.
#
Verbinde mitarbeiter und abteilung über abteilung_id.
#
SELECT m.vorname, m.nachname, a.name AS abteilung FROM mitarbeiter m INNER JOIN abteilung a ON m.abteilung_id = a.abteilung_id;

##
#
12
#
JOIN
#
Gib alle Fahrten mit Datum, Kennzeichen, Nachname des Fahrers und gefahrenen Kilometern aus.
#
Verbinde fahrt mit fahrzeug und mit mitarbeiter.
#
SELECT f.datum, fz.kennzeichen, m.nachname, f.kilometer FROM fahrt f INNER JOIN fahrzeug fz ON f.fahrzeug_id = fz.fahrzeug_id INNER JOIN mitarbeiter m ON f.mitarbeiter_id = m.mitarbeiter_id;
#
##
##
#
13
#
JOIN
#
Gib alle Wartungen mit Kennzeichen, Modell, Art und Kosten aus, sortiert nach Datum.
#
Verbinde wartung und fahrzeug.
#
SELECT w.datum, fz.kennzeichen, fz.modell, w.art, w.kosten FROM wartung w INNER JOIN fahrzeug fz ON w.fahrzeug_id = fz.fahrzeug_id ORDER BY w.datum;
#
##
##
#
14
#
JOIN
#
Gib alle Fahrzeuge mit ihren Wartungsterminen aus. Fahrzeuge ohne Wartung sollen ebenfalls erscheinen.
#
LEFT JOIN behält alle Fahrzeuge.
#
SELECT fz.kennzeichen, w.datum, w.art FROM fahrzeug fz LEFT JOIN wartung w ON fz.fahrzeug_id = w.fahrzeug_id;

15
#
JOIN
#
Welche Fahrzeuge (Kennzeichen) wurden von Mitarbeitern der Abteilung Logistik gefahren? Jedes Kennzeichen nur einmal.
#
Vier Tabellen: fahrt, fahrzeug, mitarbeiter, abteilung. Doppelte entfernst du mit DISTINCT.
#
SELECT DISTINCT fz.kennzeichen FROM fahrt f INNER JOIN fahrzeug fz ON f.fahrzeug_id = fz.fahrzeug_id INNER JOIN mitarbeiter m ON f.mitarbeiter_id = m.mitarbeiter_id INNER JOIN abteilung a ON m.abteilung_id = a.abteilung_id WHERE a.name = 'Logistik';

##
#
17
#
GRUPPIERUNG
#
Wie viele Kilometer wurden je Fahrzeug gefahren? Gib Kennzeichen und Summe aus, absteigend sortiert.
#
JOIN von fahrt und fahrzeug, danach GROUP BY Kennzeichen.
#
SELECT fz.kennzeichen, SUM(f.kilometer) AS km_gesamt FROM fahrt f INNER JOIN fahrzeug fz ON f.fahrzeug_id = fz.fahrzeug_id GROUP BY fz.kennzeichen ORDER BY km_gesamt DESC;
#
##
##
#
18
#
GRUPPIERUNG
#
Wie hoch sind die Wartungskosten je Wartungsart?
#
Gruppiere wartung nach art.
#
SELECT art, ROUND(SUM(kosten), 2) AS kosten_gesamt FROM wartung GROUP BY art;
#
##
##
#
19
#
GRUPPIERUNG
#
Für welche Fahrzeuge lagen die Wartungskosten insgesamt über 1.000 Euro? Gib die Fahrzeug-ID und die Summe aus.
#
Bedingungen auf Summen formulierst du mit HAVING.
#
SELECT fahrzeug_id, ROUND(SUM(kosten), 2) AS kosten_gesamt FROM wartung GROUP BY fahrzeug_id HAVING SUM(kosten) > 1000;
#

20
#
GRUPPIERUNG
#
Welche Mitarbeitern haben mindestens vier Fahrten unternommen? Gib Nachname und Anzahl der Fahrten aus.
#
JOIN von mitarbeiter und fahrt, GROUP BY je Person und HAVING COUNT(*) >= 4.
#
SELECT m.nachname, COUNT(*) AS anzahl_fahrten FROM mitarbeiter m INNER JOIN fahrt f ON m.mitarbeiter_id = f.mitarbeiter_id GROUP BY m.mitarbeiter_id, m.nachname HAVING COUNT(*) >= 4;
#
