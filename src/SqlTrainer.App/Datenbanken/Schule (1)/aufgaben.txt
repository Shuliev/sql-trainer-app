AGGREGAT
#
Wie lautet der Notendurchschnitt über alle vergebenen Noten, gerundet auf zwei Nachkommastellen?

Kombiniere ROUND() und AVG().

#

#
JOIN
#
Gib alle Noten mit Nachname der Schülers, Fachname, Note und Datum aus.
#
Verbinde note mit schueler und mit fach.
#
SELECT s.nachname, f.name AS fach, n.note, n.datum FROM note n INNER JOIN schueler s ON n.schueler_id = s.schueler_id INNER JOIN fach f ON n.fach_id = f.fach_id;

JOIN
#
Gib alle Fächer mit den IDs der dort vergebenen Noten aus. Fächer ohne Noten sollen ebenfalls erscheinen.
#
LEFT JOIN behält alle Fächer.
#
SELECT f.name, n.note_id FROM fach f LEFT JOIN note n ON f.fach_id = n.fach_id;

JOIN
#
Gib alle Noten der Klasse FI24A im Fach Datenbanken mit Nachname und Note aus.
#

GRUPPIERUNG
#
Berechne den Notendurchschnitt je Fach (Fachname und Durchschnitt, gerundet auf zwei Nachkommastellen).
#
JOIN zwischen note und fach, danach GROUP BY Fachname.

##
#
19
#
GRUPPIERUNG
#
Welche Schüler haben einen Notendurchschnitt besser als 3,0? Gib Nachname und Durchschnitt aus.
#
Besser heißt kleiner. Filtere die Gruppen mit HAVING AVG(...) < 3.
#
SELECT s.nachname, ROUND(AVG(n.note), 2) AS durchschnitt FROM schueler s INNER JOIN note n ON s.schueler_id = n.schueler_id GROUP BY s.schueler_id, s.nachname HAVING AVG(n.note) < 3;
#

#
20
#
GRUPPIERUNG
#
Welche Lehrer verantworten mehr als ein Fach? Gib Nachname und Anzahl der Fächer aus.
#
JOIN zwischen lehrer und fach, GROUP BY je Lehrer und HAVING COUNT(*) > 1.
#
SELECT l.nachname, COUNT(*) AS anzahl_faecher FROM lehrer l INNER JOIN fach f ON l.lehrer_id = f.lehrer_id GROUP BY l.lehrer_id, l.nachname HAVING COUNT(*) > 1;
