SELECT

11
#
JOIN
#
Gib zu jedem Produkt den Produktnamen und den Namen seiner Kategorie aus.
#
Verbinde produkt und kategorie über kategorie_id mit INNER JOIN.
#
SELECT p.name AS produkt, k.name AS kategorie FROM produkt p INNER JOIN kategorie k ON p.kategorie_id = k.kategorie_id;

#
12
#
JOIN
#
Gib alle Bestellungen mit Bestellnummer, Bestelldatum sowie Vor- und Nachname des Kunden aus.
#
Die Verknüpfung erfolgt über kunde_id.
#
SELECT b.bestellung_id, b.bestelldatum, k.vorname, k.nachname FROM bestellung b INNER JOIN kunde k ON b.kunde_id = k.kunde_id;
#
##
##
#
13
#
JOIN
#
Gib für die Bestellung mit der Nummer 1 alle Positionen mit Produktname, Menge und Einzelpreis aus.
#
Verbinde bestellposition mit produkt und filtere auf bestellung_id = 1.
#
SELECT p.name, bp.menge, bp.einzelpreis FROM bestellposition bp INNER JOIN produkt p ON bp.produkt_id = p.produkt_id WHERE bp.bestellung_id = 1;
#
##
##
#
14
#
JOIN
#
Gib alle Kunden mit ihren Bestellnummern aus. Kunden ohne Bestellung sollen ebenfalls erscheinen.
#
Ein LEFT JOIN behält alle Zeilen der linken Tabelle, auch ohne Partner.
#
SELECT k.vorname, k.nachname, b.bestellung_id FROM kunde k LEFT JOIN bestellung b ON k.kunde_id = b.kunde_id;
#
##
##
#
15
#
JOIN
#
Welche Produkte hat der Kunde Lukas Müller bestellt? Jedes Produkt soll nur einmal erscheinen.
#
Du brauchst vier Tabellen: kunde, bestellung, bestellposition, produkt. Doppelte vermeidest du mit DISTINCT.
#
SELECT DISTINCT p.name FROM kunde k INNER JOIN bestellung b ON k.kunde_id = b.kunde_id INNER JOIN bestellposition bp ON b.bestellung_id = bp.bestellung_id INNER JOIN produkt p ON bp.produkt_id = p.produkt_id WHERE k.vorname = 'Lukas' AND k.nachname = 'Müller';

GRUPPIERUNG
#
Wie viele Produkte gibt es je Kategorie? Gib die Kategorie-ID und die Anzahl aus.
#

GRUPPIERUNG
#
Berechne für jede Bestellung den Gesamtwert (Summe aus Menge mal Einzelpreis), gerundet auf zwei Nachkommastellen.
#
Gruppiere bestellposition nach bestellung_id und summiere menge * einzelpreis.

19
#
GRUPPIERUNG
#
Welche Kategorien enthalten mindestens drei Produkte? Gib den Kategorienamen und die Anzahl aus.
#
Bedingungen auf Gruppen formulierst du mit HAVING, nicht mit WHERE.
#
SELECT k.name, COUNT(*) AS anzahl FROM kategorie k INNER JOIN produkt p ON k.kategorie_id = p.kategorie_id GROUP BY k.name HAVING COUNT(*) >= 3;
#
##
##
#
20
#
GRUPPIERUNG
#
Welche Kunden haben insgesamt für mehr als 500 Euro bestellt? Gib Nachname und Umsatz aus, absteigend sortiert.
#
Verbinde kunde, bestellung und bestellposition, gruppiere je Kunde und filtere mit HAVING SUM(...) > 500.
#
SELECT k.nachname, ROUND(SUM(bp.menge * bp.einzelpreis), 2) AS umsatz FROM kunde k INNER JOIN bestellung b ON k.kunde_id = b.kunde_id INNER JOIN bestellposition bp ON b.bestellung_id = bp.bestellung_id GROUP BY k.kunde_id, k.nachname HAVING SUM(bp.menge * bp.einzelpreis) > 500 ORDER BY umsatz DESC;
#
