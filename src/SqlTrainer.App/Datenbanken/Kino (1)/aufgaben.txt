SELECT
#

11
#
JOIN
#
Gib alle Vorstellungen mit Datum, Uhrzeit und Filmtitel aus.
#
Verbinde vorstellung und film über film_id.
#
SELECT v.datum, v.uhrzeit, f.titel FROM vorstellung v INNER JOIN film f ON v.film_id = f.film_id;
#
##
##
#
12
#
JOIN
#
Gib alle Vorstellungen mit Filmtitel und Saalname aus, sortiert nach Datum und Uhrzeit.
#
Zwei JOINs: vorstellung-film und vorstellung-saal. ORDER BY mit zwei Spalten.
#
SELECT v.datum, v.uhrzeit, f.titel, s.name AS saal FROM vorstellung v INNER JOIN film f ON v.film_id = f.film_id INNER JOIN saal s ON v.saal_id = s.saal_id ORDER BY v.datum, v.uhrzeit;
#

#
Verbinde ticket und kunde über kunde_id.

JOIN
#
Gib alle Filme mit den IDs ihrer Vorstellungen aus. Filme ohne Vorstellung sollen ebenfalls erscheinen.
#
LEFT JOIN behält alle Filme.

15
#
JOIN
#
Welche Filme hat Selin Kaya gesehen? Jeder Titel nur einmal.
#
Vier Tabellen: kunde, ticket, vorstellung, film. Verwende DISTINCT.
#
SELECT DISTINCT f.titel FROM kunde k INNER JOIN ticket t ON k.kunde_id = t.kunde_id INNER JOIN vorstellung v ON t.vorstellung_id = v.vorstellung_id INNER JOIN film f ON v.film_id = f.film_id WHERE k.vorname = 'Selin' AND k.nachname = 'Kaya';

GRUPPIERUNG
#
Wie viele Tickets wurden je Vorstellung verkauft? Sortiere absteigend nach Anzahl.

##
#
18
#
GRUPPIERUNG
#
Wie hoch ist der Ticketumsatz je Film? Gib Filmtitel und Umsatz aus.
#
Der Preis steht in vorstellung. Verbinde ticket, vorstellung und film und summiere den Preis.
#
SELECT f.titel, SUM(v.preis) AS umsatz FROM ticket t INNER JOIN vorstellung v ON t.vorstellung_id = v.vorstellung_id INNER JOIN film f ON v.film_id = f.film_id GROUP BY f.titel ORDER BY umsatz DESC;

19
#
GRUPPIERUNG
#
Welche Kunden haben mehr als fünf Tickets gekauft? Gib Nachname und Anzahl aus.
#
JOIN von kunde und ticket, GROUP BY je Kunde, dann HAVING.
#
SELECT k.nachname, COUNT(*) AS anzahl FROM kunde k INNER JOIN ticket t ON k.kunde_id = t.kunde_id GROUP BY k.kunde_id, k.nachname HAVING COUNT(*) > 5;
#

20
#
GRUPPIERUNG
#
In welchen Sälen finden mindestens sechs Vorstellungen statt? Gib Saalname und Anzahl aus.
#
JOIN von saal und vorstellung, GROUP BY Saal und HAVING COUNT(*) >= 6.
#
SELECT s.name, COUNT(*) AS anzahl FROM saal s INNER JOIN vorstellung v ON s.saal_id = v.saal_id GROUP BY s.saal_id, s.name HAVING COUNT(*) >= 6;
